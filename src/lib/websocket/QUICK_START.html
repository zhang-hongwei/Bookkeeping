WebSocket 库快速开始

🚀 5分钟上手

1. 基础使用

import { useWebSocket } from '@/components/layouts/websocket';

function MyComponent() {
    const { isConnected, send, onMessage } = useWebSocket({
        config: { url: 'wss://api.example.com/ws' }
    });

    useEffect(() => {
        return onMessage((data) => console.log('收到:', data));
    }, [onMessage]);

    return (
        <div>
            <p>状态: {isConnected ? '已连接' : '未连接'}</p>
            <button onClick={() => send({ message: 'Hello' })}>
                发送消息
            </button>
        </div>
    );
}

🎯 核心 Hook

useWebSocket - 基础连接

const {
    isConnected,     // 是否连接

📊 实用示例

聊天应用

function ChatApp() {
    const [message, setMessage] = useState('');
    const { isConnected, send } = useWebSocketContext();
    const { messages } = useWebSocketMessage({
        client: null, // 从 context 获取
        filter: (data) => data.type === 'chat'
    });

    const sendMessage = () => {
        if (message.trim()) {
            send({ type: 'chat', message, user: 'me' });
            setMessage('');
        }
    };

    return (
        <div>
            <div className="messages">
                {messages.map((msg, i) => (
                    <div key={i}>{msg.user}: {msg.message}</div>
                ))}
            </div>
            <input
                value={message}
                onChange={(e) => setMessage(e.target.value)}
                onKeyPress={(e) => e.key === 'Enter' && sendMessage()}
            />
            <button onClick={sendMessage} disabled={!isConnected}>
                发送
            </button>
        </div>
    );
}

实时数据

function DataMonitor() {
    const { messages } = useWebSocketMessage({
        client: null,
        filter: (data) => data.type === 'metric',
        maxMessages: 20
    });

    const latestData = messages[messages.length - 1];

    return (
        <div>
            <h3>实时数据</h3>
            {latestData && (
                <div>
                    <p>值: {latestData.value}</p>
                    <p>时间: {new Date(latestData.timestamp).toLocaleString()}</p>
                </div>
            )}
        </div>
    );
}

通知系统

function NotificationBell() {
    const [notifications, setNotifications] = useState([]);
    const { addMessageListener } = useWebSocketContext();

    useEffect(() => {
        return addMessageListener((data) => {
            if (data.type === 'notification') {
                setNotifications(prev => [data, ...prev].slice(0, 10));
            }
        });
    }, [addMessageListener]);

    return (
        <div className="notification-bell">
            🔔 {notifications.length > 0 && (
                <span className="badge">{notifications.length}</span>
            )}
        </div>
    );
}

💡 最佳实践

✅ 推荐做法

// 1. 使用 Provider 进行全局管理
<WebSocketProvider config={{ url: 'wss://api.example.com/ws' }}>
    <App />
</WebSocketProvider>

// 2. 正确清理监听器
useEffect(() => {
    const unsubscribe = onMessage(handleMessage);
    return unsubscribe; // 自动清理
}, [onMessage]);

// 3. 使用错误边界
<WebSocketErrorBoundary>
    <WebSocketProvider>
        <App />
    </WebSocketProvider>
</WebSocketErrorBoundary>

// 4. 限制消息数量
const { messages } = useWebSocketMessage({
    client,
    maxMessages: 100, // 防止内存泄漏
    enableHistory: false // 如果不需要历史记录
});

❌ 避免做法

// ❌ 不要在每次渲染时创建新连接
function BadComponent() {
    const ws = useWebSocket({ config: { url: 'wss://...' } }); // 每次都创建新连接
}

// ❌ 不要忘记清理监听器
useEffect(() => {
    onMessage(handleMessage); // 没有返回清理函数
}, []);

// ❌ 不要无限制地存储消息
const { messages } = useWebSocketMessage({
    client,
    // 没有设置 maxMessages，可能导致内存泄漏
});
